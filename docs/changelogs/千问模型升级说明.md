```python
def call_qwen_vision(prompt, image, stream=False, model_id='qwen3.5-plus'):
```

```python
def call_vision_model(prompt, image, model_type='qwen', stream=False):

        # 自动获取对应的model_id
        model_id = AVAILABLE_MODELS[model_type]['model_id']
```
